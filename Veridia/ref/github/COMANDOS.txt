=== 3. Abertura das árvores (sem blobs, sem checkout) ===
Script open_tree.sh:
#!/usr/bin/env bash
# Abre a árvore de um repositório público via git (clone parcial sem blobs, sem checkout, profundidade 1).
# Uso: open_tree.sh owner/repo
set -u
repo="$1"
name="${repo/\//__}"
base=/home/user/survey
d="$base/trees/$name.git"
out="$base/meta/$name.txt"
[ -s "$out" ] && grep -q '^status=ok' "$out" && exit 0
rm -rf "$d"
start=$(date +%s)
if GIT_LFS_SKIP_SMUDGE=1 timeout 600 git clone --quiet --bare --filter=blob:none --depth 1 "https://github.com/$repo" "$d" 2> "$base/meta/$name.err"; then
  {
    echo "status=ok"
    echo "repo=$repo"
    echo "collected_at=$(date -u +%Y-%m-%dT%H:%M:%SZ)"
    echo "default_branch=$(git -C "$d" symbolic-ref --short HEAD 2>/dev/null)"
    git -C "$d" log -1 --format='head_commit=%H%nhead_date=%cI%nhead_subject=%s' HEAD
    echo "n_files=$(git -C "$d" ls-tree -r --name-only HEAD | wc -l)"
    echo "clone_secs=$(( $(date +%s) - start ))"
  } > "$out"
  git -C "$d" ls-tree -r --name-only HEAD > "$base/trees/$name.files"
else
  { echo "status=fail"; echo "repo=$repo"; echo "collected_at=$(date -u +%Y-%m-%dT%H:%M:%SZ)"; echo "error=$(tr '\n' ' ' < "$base/meta/$name.err" | cut -c1-300)"; } > "$out"
fi
